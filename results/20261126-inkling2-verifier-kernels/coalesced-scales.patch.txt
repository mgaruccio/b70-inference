+"""Measure scale prepacking and compare isolated NVFP4 Xe2 GEMMs on B70.
+
+After rebuilding the extension with scripts/build-cutlass-xe2.sh, run:
+    python scripts/bench-nvfp4-scale-layout.py --tokens 1 8 32
+
+The shapes are from Inkling 2 Mini; inputs and routes are synthetic. Report median
+XPU event times, excluding host transfers and routing. Both layouts must produce
+bitwise-identical BF16 outputs. This is not a serving benchmark.
+"""
+
+import argparse
+from functools import partial
+from statistics import median
+
+
+def _measure(calls, iterations, *, before=None):
+    import torch
+
+    start = torch.xpu.Event(enable_timing=True)
+    end = torch.xpu.Event(enable_timing=True)
+    names = tuple(calls)
+    samples = {name: [] for name in names}
+    for i in range(5 + iterations):
+        # Alternate A/B order to reduce bias from clocks and warm caches.
+        for name in (names if i % 2 == 0 else reversed(names)):
+            if before is not None:
+                before()
+            start.record()
+            result = calls[name]()
+            end.record()
+            end.synchronize()
+            if i >= 5:
+                samples[name].append(start.elapsed_time(end))
+            del result
+    return {name: median(times) for name, times in samples.items()}
+
+
+def main():
+    parser = argparse.ArgumentParser(
+        description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter
+    )
+    parser.add_argument("--tokens", type=int, nargs="+", default=[1, 8, 32])
+    parser.add_argument("--iterations", type=int, default=40)
+    args = parser.parse_args()
+    if min(args.tokens) <= 0 or args.iterations <= 0:
+        parser.error("tokens and iterations must be positive")
+
+    import torch
+
+    from vllm_inkling2_xpu.ops.moe_cutlass import nvfp4_ext, prepack_nvfp4_scales
+
+    torch.manual_seed(1039)
+    device = "xpu"
+    e, hidden, inter, topk = 128, 2304, 1152, 8
+    outer = torch.linspace(0.02, 0.04, e, dtype=torch.float32, device=device)
+    atomic = torch.zeros(1, dtype=torch.int32, device=device)
+    kernels = {
+        "stock": nvfp4_ext.nvfp4_grouped_gemm_xe2,
+        "coalesced": nvfp4_ext.nvfp4_grouped_gemm_xe2_coalesced_scales,
+    }
+    for proj, n, k in (("w13", 2 * inter, hidden), ("w2", hidden, inter)):
+        weight = torch.randint(256, (e, n, k // 2), dtype=torch.uint8, device=device)
+        scales = torch.randint(0x7F, (e, n, k // 16), dtype=torch.uint8, device=device)
+        stock = scales.view(torch.float8_e4m3fn)
+        pack = partial(prepack_nvfp4_scales, stock)
+        pack_ms = _measure({"pack": pack}, args.iterations)["pack"]
+        print(f"{proj}: prepack {stock.numel() / 2**20:.2f} MiB, {pack_ms:.3f} ms")
+        layouts = {"stock": stock, "coalesced": pack()}
+        for tokens in args.tokens:
+            ids = torch.rand(tokens, e).argsort(dim=1)[:, :topk]
+            counts = ids.flatten().bincount(minlength=e).to(
+                device=device, dtype=torch.int32
+            )
+            a = torch.randn(tokens * topk, k, dtype=torch.bfloat16, device=device)
+            out = {name: a.new_empty((tokens * topk, n)) for name in kernels}
+            calls = {
+                name: partial(
+                    gemm, a, weight, layouts[name], outer, out[name], counts,
+                    n, k, e, atomic,
+                )
+                for name, gemm in kernels.items()
+            }
+            # Each persistent GEMM consumes a fresh counter; keep reset untimed.
+            ms = _measure(calls, args.iterations, before=atomic.zero_)
+            stock_bits = out["stock"].view(torch.int16)
+            coalesced_bits = out["coalesced"].view(torch.int16)
+            if not torch.equal(stock_bits, coalesced_bits):
+                raise RuntimeError(f"{proj} tokens={tokens}: BF16 outputs differ")
+            print(
+                f"  tokens={tokens:3}: stock {ms['stock'] * 1000:8.1f} us, "
+                f"coalesced {ms['coalesced'] * 1000:8.1f} us, "
+                f"{ms['stock'] / ms['coalesced']:.3f}x, bitwise equal"
+            )
+
+
+if __name__ == "__main__":
+    main()

+def prepack_nvfp4_scales(scales: torch.Tensor) -> torch.Tensor:
+    """Physically transpose [E, N, K/16] E4M3 bytes into [E, K/32, N, 2]."""
+    if (
+        scales.ndim != 3
+        or scales.size(-1) % 2
+        or scales.dtype != torch.float8_e4m3fn
+        or not scales.is_contiguous()
+    ):
+        raise ValueError("Expected contiguous NVFP4 E4M3 scales with complete pairs")
+    experts, n, groups = scales.shape
+    # Copy bytes, not FP8 values; keep the low/high scale within each K tile.
+    pairs = scales.view(torch.uint8).view(experts, n, groups // 2, 2)
+    return pairs.transpose(1, 2).contiguous().view(scales.dtype)
